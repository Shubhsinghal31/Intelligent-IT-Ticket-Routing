# Authentication & Security
# ============================================
passlib[bcrypt]==1.7.4
# Pinned explicitly, not left to passlib's [bcrypt] extra to resolve.
# passlib 1.7.4 (unmaintained since ~2020) reads bcrypt.__about__.__version__
# to detect the backend version — bcrypt removed that module in 4.1.0
# (Oct 2023). Any bcrypt >=4.1 breaks passlib's version probe, which
# then raises inside verify_password() and gets treated as "wrong
# password" for every login, regardless of whether the password is
# correct. Unpinned, `pip install` resolves whatever bcrypt is newest
# at install time — a .venv created before Oct 2023 would silently get
# a working version, while a fresh install (e.g. every `docker build`)
# gets the latest, broken one. Pinning here makes every install
# (native or Docker) resolve the same known-good version instead of
# depending on when `pip install` happened to run.
bcrypt==4.0.1
python-jose[cryptography]==3.3.0
